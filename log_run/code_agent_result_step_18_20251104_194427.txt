def extract_author_info(pdf_file):
    try:
        with open(pdf_file, 'rb') as f:
            pdf_reader = PyPDF2.PdfReader(f)
            text = ''
            for page in pdf_reader.pages[:2]:  # Check the first two pages
                text += page.extract_text()
            lines = text.split('\n')
            
            author_name = None
            author_email = None
            author_affiliation = None
            
            for line in lines:
                if author_name is None and re.match(r'^[A-Z][a-z]+ [A-Z][a-z]+', line.strip()):
                    author_name = line.strip()
                if '@' in line:
                    author_email = line.strip()
                if 'University' in line or 'Institute' in line or 'Department' in line:
                    author_affiliation = line.strip()
                    
            return author_name, author_email, author_affiliation
    
    except Exception as e:
        print(f"Error processing {pdf_file}: {e}")
        return None, None, None
