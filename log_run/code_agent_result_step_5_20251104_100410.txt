def calculate_period_rate():

            (16.268, 7),
            (10.478, 2),
            (18.584, 7)
        ]
        
        # Write data to worksheet
        ws.cell(row=1, column=1, value='Annual Rate (%*Period per year)')
        ws.cell(row=1, column=2, value='Period per year')
        ws.cell(row=1, column=3, value='Period Rate (%)')
        
        for i, (annual_rate, period_per_year) in enumerate(data, start=2):
            ws.cell(row=i, column=1, value=annual_rate)
            ws.cell(row=i, column=2, value=period_per_year)
            period_rate = annual_rate / period_per_year
            ws.cell(row=i, column=3, value=period_rate)
        
        # Convert results to number type
        for row in ws.iter_rows(min_row=2, min_col=3, max_col=3):
            for cell in row:
                cell.number_format = '0.00%'
        
        # Find and highlight the highest result
        max_period_rate = max((cell.value for row in ws.iter_rows(min_row=2, min_col=3, max_col=3) for cell in row))
        for row in ws.iter_rows(min_row=2, min_col=3, max_col=3):
            for cell in row:
                if cell.value == max_period_rate:
                    cell.font = Font(color='00ff00')  # Green color
        
        # Save the workbook
        wb.save('PeriodRate.xlsx')
