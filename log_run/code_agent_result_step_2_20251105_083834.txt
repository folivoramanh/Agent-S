import os
import PyPDF2

def get_page_number(pdf_reader, item):
    if isinstance(item, dict) and '/Page' in item:
        return pdf_reader.get_destination_page_number(item)
    elif isinstance(item, list):
        for sub_item in item:
            page_num = get_page_number(pdf_reader, sub_item)
            if page_num is not None:
                return page_num
    return None

# Open the PDF file
pdf_path = os.path.expanduser("~/Desktop/book/Spectral Graph Theory.pdf")
with open(pdf_path, 'rb') as file:
    pdf_reader = PyPDF2.PdfReader(file)
    
    # Extract the outline
    outline = pdf_reader.outline
    
    # Iterate through the outline and split the PDF
    for i, item in enumerate(outline):
        if isinstance(item, dict):
            title = item.get('/Title', f'Chapter {i+1}')
            page_num = get_page_number(pdf_reader, item)
            
            # Determine the end page of the current chapter
            if i < len(outline) - 1:
                next_item = outline[i+1]
                next_page_num = get_page_number(pdf_reader, next_item)
            else:
                next_page_num = len(pdf_reader.pages)
            
            # Create a new PDF writer for the current chapter
            chapter_writer = PyPDF2.PdfWriter()
            
            # Add pages to the chapter writer
            for page in range(page_num, next_page_num):
                chapter_writer.add_page(pdf_reader.pages[page])
            
            # Save the chapter to a new PDF file
            title = title.replace(':', '').replace('/', '-').strip()
            chapter_path = os.path.expanduser(f"~/Desktop/book/{title}.pdf")
            with open(chapter_path, 'wb') as chapter_file:
                chapter_writer.write(chapter_file)
            
            print(f"Created: {title}.pdf")
