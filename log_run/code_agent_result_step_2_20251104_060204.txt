import pyexcel_ods3

def process_ods_file(file_path):
    data = pyexcel_ods3.get_data(file_path)
    if data:
        sheet_name = list(data.keys())[0]  
        sheet_data = data[sheet_name]

        processed_data = []
        for row in sheet_data[1:]:  
            if row:
                cell_value = row[0]
                if cell_value:
                    parts = str(cell_value).split()
                    first_name = parts[0]
                    last_name = ' '.join(parts[1:-1]) if len(parts) > 2 else ''
                    rank = parts[-1]
                    processed_data.append([first_name, last_name, rank])
                else:
                    processed_data.append(['', '', ''])
            else:
                processed_data.append(['', '', ''])

        for i, row_data in enumerate(processed_data, start=1):
            if len(sheet_data[i]) >= 1:  
                sheet_data[i] = sheet_data[i][:1] + row_data + sheet_data[i][4:]  
            else:
                sheet_data[i] = [''] + row_data  

        pyexcel_ods3.save_data(file_path, data)
        
        # Print the modified file content
        print(pyexcel_ods3.get_data(file_path))

        print("Failed to load data from the .ods file.")
