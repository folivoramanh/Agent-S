# Load the workbook
wb = openpyxl.load_workbook('Create_column_charts_using_statistics.xlsx')
sheet = wb['Sheet1']

# Function to calculate total costs for each month in a given range
def calculate_total_costs(start_row, end_row):
    total_costs = []
    for row in range(start_row, end_row + 1):
        total_cost = sum([sheet.cell(row=row, column=col).value for col in range(2, 9)])
        total_costs.append(total_cost)
    return total_costs

# Calculate total costs for 2019 and 2020
total_costs_2019 = calculate_total_costs(3, 14)
total_costs_2020 = calculate_total_costs(20, 31)

# Print the total costs for verification
