import requests
from bs4 import BeautifulSoup

# List of years to search for
years = [2022, 2021, 2020, 2019]

# Initialize an empty list to store the results
results = []

for year in years:
    url = f"https://aclanthology.org/events/acl-{year}/"
    response = requests.get(url)
    
    if response.status_code == 200:
        soup = BeautifulSoup(response.content, 'html.parser')
        
        # Search for the best paper awards
        best_paper_awards = soup.find_all('strong', text=lambda t: t and 'Best Long Paper' in t)
        
        for award in best_paper_awards:
            title = award.parent.find_next_sibling('span').text.strip()
            authors = award.parent.find_next_sibling('span').find_next_sibling('span').text.strip()
            pdf_link = award.parent.find_next_sibling('a', href=True)['href']
            results.append({
                'title': title,
                'year': year,
                'authors': authors,
                'pdf_link': pdf_link
            })
    else:
        print(f"Failed to retrieve the ACL {year} page")

# Print the results
for result in results:
    print(result)
