    today = datetime.today()
    age = today.year - dob.year - ((today.month, today.day) < (dob.month, dob.day))
    return age

# Iterate through rows to calculate age and update the 'Age' column
for row in range(2, sheet.max_row + 1):
    dob_cell = sheet.cell(row=row, column=3).value
    if dob_cell:
        if isinstance(dob_cell, datetime):
            dob = dob_cell
        else:
            try:
                dob = datetime.strptime(dob_cell, '%m/%d/%Y')
            except ValueError:
                print(f"Row {row}: Invalid DOB format - {dob_cell}")
                continue
        age = calculate_age(dob)
        sheet.cell(row=row, column=4).value = age  # Update the 'Age' column
