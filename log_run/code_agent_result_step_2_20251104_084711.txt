```python
from openpyxl import load_workbook
from datetime import datetime

# Load the workbook
wb = load_workbook(filename='Employee_Age_By_Birthday.xlsx')
sheet = wb['Sheet1']

# Function to calculate age from date of birth
def calculate_age(dob):
    today = datetime.today()
    age = today.year - dob.year - ((today.month, today.day) < (dob.month, dob.day))
    return age

# Iterate through rows and calculate age
for row in range(2, sheet.max_row + 1):
    dob_str = sheet[f'C{row}'].value
    if dob_str and isinstance(dob_str, str):
        try:
            dob = datetime.strptime(dob_str, '%m/%d/%Y')
            age = calculate_age(dob)
            sheet[f'D{row}'].value = age
        except ValueError:
            # Handle invalid date formats
            sheet[f'D{row}'].value = 'Invalid Date'
    else:
        sheet[f'D{row}'].value = 'No DOB'

# Save the workbook
wb.save('Employee_Age_By_Birthday.xlsx')
```
