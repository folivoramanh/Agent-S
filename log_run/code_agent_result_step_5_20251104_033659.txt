```python
import pandas as pd

# Load the Excel file
file_path = 'NetIncome.xlsx'
try:
    # Read Sheet1
    sheet1_data = pd.read_excel(file_path, sheet_name='Sheet1')
    
    # Calculate the sum of 'Revenue' and 'Total Expenses'
    total_revenue = sheet1_data['Revenue'].sum()
    total_expenses = sheet1_data['Total Expenses'].sum()
    
    # Create a DataFrame for the results
    results = pd.DataFrame({'Total Revenue': [total_revenue], 'Total Expenses': [total_expenses]})
    
    # Write the results to Sheet2
    with pd.ExcelWriter(file_path, mode='a', if_sheet_exists='replace', engine='openpyxl') as writer:
        results.to_excel(writer, sheet_name='Sheet2', index=False)
        
    print("Results written to Sheet2 successfully.")
except FileNotFoundError:
    print("File not found.")
except Exception as e:
    print(f"An error occurred: {e}")
```
