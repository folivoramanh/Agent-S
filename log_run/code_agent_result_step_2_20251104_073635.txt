from openpyxl import load_workbook
from openpyxl.chart import LineChart, Reference

# Load the workbook
wb = load_workbook(filename='SalesRep.xlsx')
sheet = wb.active

# Calculate month-on-month growth percentage
totals = [cell.value for cell in sheet[12]][1:]  # Get total sales values
growth_percentages = []
for i in range(1, len(totals)):
    growth = (totals[i] - totals[i-1]) / totals[i-1] * 100
    growth_percentages.append(growth)

# Add the "Growth" row
sheet.append(["Growth"] + [None] + growth_percentages)

# Create a line chart for growth percentages
chart = LineChart()
chart.title = "Growth"
chart.y_axis.title = 'Growth %'
chart.x_axis.title = 'Month'

data = Reference(sheet, min_col=3, max_col=7, min_row=13, max_row=13)  # Growth row
cats = Reference(sheet, min_col=3, max_col=7, min_row=1, max_row=1)  # Month headers

chart.add_data(data, titles_from_data=False)
chart.set_categories(cats)

sheet.add_chart(chart, "I15")

# Save the workbook
wb.save('SalesRep.xlsx')

# Print the updated state of the sheet
for row in sheet.iter_rows(values_only=True):
    print(row)
