```python
import pandas as pd

# Load the 'unseen_movies' sheet from the Excel file
file_path = 'movie_list.xlsx'
try:
    unseen_movies_data = pd.read_excel(file_path, sheet_name='unseen_movies')
    
    # Rename the columns for clarity
    unseen_movies_data.columns = ['Title', 'Year', 'IMDB Rating']
    
    # Filter movies released before 2024
    filtered_data = unseen_movies_data[unseen_movies_data['Year'] < 2024]
    
    # Sort the filtered data by IMDB Rating in descending order
    sorted_data = filtered_data.sort_values(by='IMDB Rating', ascending=False)
    
    # Write the sorted data to a new sheet 'filtered_movies'
    with pd.ExcelWriter(file_path, mode='a', if_sheet_exists='replace') as writer:
        sorted_data.to_excel(writer, sheet_name='filtered_movies', index=False)
    print("Data has been written to 'filtered_movies' sheet.")
except Exception as e:
    print("An error occurred:", str(e))
```
