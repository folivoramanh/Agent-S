projects_path = os.path.join(desktop_path, 'Projects')
miscellaneous_path = os.path.join(desktop_path, 'Miscellaneous')

# Create paths if they don't exist
for path in [paper_reading_path, projects_path, miscellaneous_path]:
    if not os.path.exists(path):
        os.makedirs(path)

# List files on desktop
files = [f for f in os.listdir(desktop_path) if os.path.isfile(os.path.join(desktop_path, f))]

for file in files:
    file_path = os.path.join(desktop_path, file)
    if file.endswith(('.pdf', '.doc', '.docx')):  # Academic papers
        shutil.move(file_path, paper_reading_path)
    elif file.endswith(('.py', '.java', '.cpp')):  # Coding projects
        shutil.move(file_path, projects_path)
    else:  # Miscellaneous
        shutil.move(file_path, miscellaneous_path)

print("Files have been organized.")
