photos_path = os.path.join(desktop_path, 'photos')
cpjpg_path = os.path.join(desktop_path, 'cpjpg')

if not os.path.exists(photos_path):
    print("The 'photos' directory does not exist on the desktop.")
else:
    # Create 'cpjpg' directory if it doesn't exist
    if not os.path.exists(cpjpg_path):
        os.makedirs(cpjpg_path)

    # Traverse 'photos' directory and copy .jpg files to 'cpjpg'
    for root, dirs, files in os.walk(photos_path):
        for file in files:
            if file.lower().endswith('.jpg'):
                file_path = os.path.join(root, file)
                shutil.copy(file_path, cpjpg_path)
                print(f"Copied {file_path} to {cpjpg_path}")
