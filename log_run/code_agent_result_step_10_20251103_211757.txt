import os
import pandas as pd

# Load the bookkeeping sheet
file_path = "my_bookkeeping.xlsx"
df = pd.read_excel(file_path)

# Parse the transaction files
transactions_path = "/home/user/Downloads/recent_transactions"
for filename in os.listdir(transactions_path):
    with open(os.path.join(transactions_path, filename), 'r') as f:
        lines = f.readlines()
        description = lines[0].strip().split(': ')[1]
        category = lines[1].strip().split(': ')[1]
        type = lines[2].strip().split(': ')[1]
        amount = float(lines[3].strip().split(': ')[1])
        
        # Create a new row
        new_row = pd.DataFrame([{
            'Description': description,
            'Category': category,
            'Type': type,
            'Amount': amount,
            'Balance': df['Balance'].iloc[-1] + amount if not df.empty else amount
        }])
        
        # Append the new row
        df = pd.concat([df, new_row], ignore_index=True)

# Update the Balance column
df['Balance'] = df['Amount'].cumsum()

# Save the updated dataframe to the Excel file
df.to_excel(file_path, index=False)

print(df)
