```python
import openpyxl

# Load the workbook
wb = openpyxl.load_workbook('SalesRep.xlsx')

# Select 'Sheet1' and 'Sheet2'
sheet1 = wb['Sheet1']
sheet2 = wb['Sheet2']

# Initialize variables to store total sales for each month
total_sales = [0] * 6  # For Jan to Jun

# Iterate through the rows in 'Sheet1' to calculate total sales
for row in sheet1.iter_rows(min_row=2, values_only=True):
    if isinstance(row[0], str) and row[0] not in ['January', 'February', 'March']:
        continue  # Skip rows that don't represent months
    for i in range(1, 7):  # Columns 'Sales' to 'Jun'
        if row[i] is not None:
            total_sales[i-1] += row[i]

# Write the total sales to 'Sheet2'
sheet2['A1'] = 'Month'
sheet2['B1'] = 'Total Sales'
months = ['January', 'February', 'March', 'April', 'May', 'June']
for i, total in enumerate(total_sales):
    sheet2[f'A{i+2}'] = months[i]
    sheet2[f'B{i+2}'] = total

# Save the workbook
wb.save('SalesRep.xlsx')
```
