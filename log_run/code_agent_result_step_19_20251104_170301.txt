import os

thunderbird_dir = "/home/user/.thunderbird"
for profile in os.listdir(thunderbird_dir):
    if profile.endswith(".default"):
        mab_path = os.path.join(thunderbird_dir, profile, 'abook.mab')
        if os.path.exists(mab_path):
            print(f"Found abook.mab at: {mab_path}")
            csv_path = "/home/user/Desktop/contacts.csv"
            
            with open(mab_path, 'r', encoding='latin1') as mab_file:
                data = mab_file.read()

            # Enhanced parsing logic for Mork format
            records = data.split('@<Card:')
            contacts = []
            for record in records[1:]:
                try:
                    name = re.search(r'FN:(.*?)\@(.*?)@', record).group(1).strip()
                    email = re.search(r'EmailAddress:(.*?)\@(.*?)@', record).group(1).strip()
                    contacts.append((name, email))
                except AttributeError:
                    continue

            with open(csv_path, 'w') as csv_file:
                csv_file.write("Name,Email\n")
                for name, email in contacts:
                    csv_file.write(f"{name},{email}\n")

            print(f"CSV file created at: {csv_path}")

            with open(csv_path, 'r') as csv_file:
                print(csv_file.read())
