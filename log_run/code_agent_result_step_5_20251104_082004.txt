```python
import openpyxl
from openpyxl import Workbook

# Load the workbook
wb = openpyxl.load_workbook('Students_Class_Subject_Marks.xlsx')
sheet = wb['Sheet1']

# Read data from B2:F5
data = []
for row in sheet['B2:F5']:
    row_data = [cell.value for cell in row]
    data.append(row_data)

# Transpose the data
transposed_data = list(map(list, zip(*data)))

# Write transposed data to B8:E12
for i, row in enumerate(transposed_data, start=8):
    for j, value in enumerate(row, start=2):
        sheet.cell(row=i, column=j).value = value

# Save the workbook
wb.save('Students_Class_Subject_Marks.xlsx')
```
