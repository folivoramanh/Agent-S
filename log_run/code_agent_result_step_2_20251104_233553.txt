from PyPDF2 import PdfReader
import os
import pandas as pd

def extract_data_from_pdf(file_path):
    try:
        with open(file_path, 'rb') as f:
            reader = PdfReader(f)
            text = reader.pages[-1].extract_text()  # Extract text from the last page
            lines = text.split('\n')
            total_line = lines[-2]  # Assuming the second last line contains the total
            parts = total_line.split()
            total_applications = parts[-9]  # Total applications
            supported_applications = parts[-5]  # Total supported applications
            return int(total_applications.replace(',', '')), int(supported_applications.replace(',', ''))
    except Exception as e:
        print(f"Error processing {file_path}: {e}")
        return None, None

directory_path = '/home/user/Documents/Fundings/ecs'
pdf_files = [f for f in os.listdir(directory_path) if f.endswith('.pdf')]

data = []
for pdf_file in pdf_files:
    file_path = os.path.join(directory_path, pdf_file)

    if total_applications is not None and supported_applications is not None:

from PyPDF2 import PdfReader
import os
import pandas as pd

def extract_data_from_pdf(file_path):
    try:
        with open(file_path, 'rb') as f:
            reader = PdfReader(f)
            text = reader.pages[-1].extract_text()
            lines = text.split('\n')
            for line in reversed(lines):
                if 'Total' in line:
                    parts = line.split()
                    total_applications = parts[-9].replace(',', '')
                    supported_applications = parts[-5].replace(',', '')
                    return int(total_applications), int(supported_applications)
    except Exception as e:
        print(f"Error processing {file_path}: {e}")
    return None, None

directory_path = '/home/user/Documents/Fundings/ecs'
pdf_files = [f for f in os.listdir(directory_path) if f.endswith('.pdf')]

data = []
for pdf_file in pdf_files:
    file_path = os.path.join(directory_path, pdf_file)
    year = 2000 + int(pdf_file[3:5])
    total_applications, supported_applications = extract_data_from_pdf(file_path)
    if total_applications is not None and supported_applications is not None:
        pass_rate = (supported_applications / total_applications) * 100
        data.append({'Year': year, 'Total Applications': total_applications, 'Supported Applications': supported_applications, 'Pass Rate': pass_rate})

df = pd.DataFrame(data)
print(df)
